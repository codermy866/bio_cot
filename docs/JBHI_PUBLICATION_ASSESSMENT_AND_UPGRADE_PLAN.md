```python
class EnsembleModel(nn.Module):
    """
    集成多个模型
    - Swin-T (权重: 0.4)
    - CNN (权重: 0.3)
    - VMamba (权重: 0.3)
    """
    def __init__(self):
        self.swin_model = load_swin_model()
        self.cnn_model = load_cnn_model()
        self.vmamba_model = load_vmamba_model()
        self.weights = [0.4, 0.3, 0.3]
    
    def forward(self, oct_images, col_images, clinical_features):
        swin_pred = self.swin_model(oct_images, col_images, clinical_features)
        cnn_pred = self.cnn_model(oct_images, col_images, clinical_features)
        vmamba_pred = self.vmamba_model(oct_images, col_images, clinical_features)
        
        # 加权平均
        ensemble_pred = (
            self.weights[0] * swin_pred +
            self.weights[1] * cnn_pred +
            self.weights[2] * vmamba_pred
        )
        return ensemble_pred
```
